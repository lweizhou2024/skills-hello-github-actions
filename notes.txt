function TRout = closeCurvedBlockSTL(curvedStl, outStl, yBack, tol)
%CLOSECURVEDBLOCKSTL Close a finely meshed curved -Y face with five coarse planes.
%
% Assumptions
% ----------
% 1. curvedStl contains ONLY the curved face.
% 2. The curved face replaces the negative-Y face of an otherwise
%    axis-aligned rectangular block.
% 3. Its boundary lies on:
%       x = xMin, x = xMax, z = zMin, z = zMax
% 4. The opposite flat face is y = yBack, with yBack greater than all
%    Y coordinates on the curved surface.
%
% Usage
% -----
% TR = closeCurvedBlockSTL("curved_face.stl", ...
%                          "closed_block.stl", ...
%                          25.0, 1e-7);
%
% The four side faces use only:
%   - the existing fine boundary vertices of the curved face, and
%   - the four corners on y = yBack.
% No transition rows or extra near-boundary vertices are created.

    arguments
        curvedStl (1,1) string
        outStl    (1,1) string
        yBack     (1,1) double
        tol       (1,1) double = 1e-7
    end

    %% Read the fine curved surface
    TRc = stlread(curvedStl);
    Vc = TRc.Points;
    Fc = TRc.ConnectivityList;

    if yBack <= max(Vc(:,2)) + tol
        error("yBack must be greater than every Y coordinate of the curved surface.");
    end

    %% Make curved-face normals point generally toward -Y
    Fc = orientTriangles(Vc, Fc, [0 -1 0]);

    %% Find and order the single outside boundary loop
    boundaryEdges = freeBoundary(triangulation(Fc, Vc));
    loop = orderSingleBoundaryLoop(boundaryEdges);
    Vloop = Vc(loop,:);

    xMin = min(Vloop(:,1));
    xMax = max(Vloop(:,1));
    zMin = min(Vloop(:,3));
    zMax = max(Vloop(:,3));

    scale = max([xMax-xMin, yBack-min(Vc(:,2)), zMax-zMin, 1]);
    planeTol = max(tol, 1e-9*scale);

    %% Start with the curved surface
    Vall = Vc;
    Fall = Fc;

    %% Add the four axis-aligned side faces
    % x = xMin, outward normal -X
    [Vf,Ff] = makeSideFace(Vc, loop, ...
        abs(Vloop(:,1)-xMin) <= planeTol, yBack, 1, [-1 0 0]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    % x = xMax, outward normal +X
    [Vf,Ff] = makeSideFace(Vc, loop, ...
        abs(Vloop(:,1)-xMax) <= planeTol, yBack, 1, [1 0 0]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    % z = zMin, outward normal -Z
    [Vf,Ff] = makeSideFace(Vc, loop, ...
        abs(Vloop(:,3)-zMin) <= planeTol, yBack, 3, [0 0 -1]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    % z = zMax, outward normal +Z
    [Vf,Ff] = makeSideFace(Vc, loop, ...
        abs(Vloop(:,3)-zMax) <= planeTol, yBack, 3, [0 0 1]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    %% Add the coarse +Y back face using only two triangles
    Vback = [xMin yBack zMin;
             xMax yBack zMin;
             xMax yBack zMax;
             xMin yBack zMax];

    Fback = [1 2 3;
             1 3 4];
    Fback = orientTriangles(Vback, Fback, [0 1 0]);

    [Vall,Fall] = appendMesh(Vall,Fall,Vback,Fback);

    %% Merge coincident vertices from independently constructed faces
    [Vout,~,map] = uniquetol(Vall, tol, ...
        "ByRows", true, "DataScale", 1);
    Fout = map(Fall);

    % Remove any triangles collapsed by vertex merging.
    keep = Fout(:,1) ~= Fout(:,2) & ...
           Fout(:,2) ~= Fout(:,3) & ...
           Fout(:,3) ~= Fout(:,1);
    Fout = Fout(keep,:);

    %% Remove duplicate triangles, independent of orientation
    Fs = sort(Fout,2);
    [~,ia] = unique(Fs,"rows","stable");
    Fout = Fout(ia,:);

    TRout = triangulation(Fout,Vout);

    %% Watertightness check: every undirected edge must occur exactly twice
    E = [Fout(:,[1 2]);
         Fout(:,[2 3]);
         Fout(:,[3 1])];
    E = sort(E,2);
    [~,~,ie] = unique(E,"rows");
    edgeUse = accumarray(ie,1);

    nOpen = nnz(edgeUse == 1);
    nNonmanifold = nnz(edgeUse > 2);

    fprintf("Output vertices:  %d\n",size(Vout,1));
    fprintf("Output triangles: %d\n",size(Fout,1));
    fprintf("Open edges:       %d\n",nOpen);
    fprintf("Nonmanifold edges:%d\n",nNonmanifold);

    if nOpen ~= 0 || nNonmanifold ~= 0
        warning("The result is not watertight. Check the curved-face boundary and tolerance.");
    end

    %% Write binary STL
    stlwrite(TRout,outStl);   % MATLAB writes binary STL by default

    %% Plot for inspection
    figure;
    trisurf(TRout, ...
        "FaceColor",[0.80 0.85 0.90], ...
        "EdgeColor",[0.20 0.20 0.20]);
    axis equal;
    xlabel("X"); ylabel("Y"); zlabel("Z");
    title("Closed STL surface mesh");
    view(3);
end


function [Vface,Fface] = makeSideFace(Vc, loop, mask, yBack, constantAxis, desiredNormal)
% Extract one contiguous curved-boundary chain and connect it directly to
% the two corresponding back-face corners.
%
% This routine deliberately avoids POLYSHAPE and DELAUNAY triangulation.
% It constructs a fan using every curved-boundary vertex, so a boundary
% whose Y coordinate varies while X or Z remains constant cannot be
% simplified down to only a few triangles.

    run = longestCircularRun(mask);

    if numel(run) < 2
        error("Could not identify a side-boundary chain.");
    end

    chainIds = loop(run);
    chain = Vc(chainIds,:);

    % Remove only repeated consecutive 3-D points. Do not simplify
    % collinear points because all curved-face boundary vertices must be
    % retained in the watertight STL.
    localScale = max(max(abs(chain),[],"all"),1);
    d = vecnorm(diff(chain,1,1),2,2);
    chain = chain([true; d > 10*eps(localScale)],:);

    if size(chain,1) < 2
        error("The extracted side boundary has fewer than two distinct points.");
    end

    % Enforce the nominal constant coordinate only within numerical noise.
    % The Y coordinate is intentionally allowed to vary along the chain.
    switch constantAxis
        case 1
            chain(:,1) = mean(chain(:,1));
        case 3
            chain(:,3) = mean(chain(:,3));
        otherwise
            error("constantAxis must be 1 or 3.");
    end

    pStart = chain(1,:);
    pEnd   = chain(end,:);

    qEnd = pEnd;
    qEnd(2) = yBack;

    qStart = pStart;
    qStart(2) = yBack;

    % Vertices are ordered as:
    %   curved chain p1 ... pn, back corner at pn, back corner at p1.
    Vface = [chain;
             qEnd;
             qStart];

    n = size(chain,1);
    iQEnd   = n + 1;
    iQStart = n + 2;

    % Fan from qStart. This creates one triangle for every consecutive pair
    % of curved-boundary points, plus one triangle along the back edge.
    % Therefore n curved-edge points produce exactly n side triangles.
    Fface = zeros(n,3);
    for i = 1:n-1
        Fface(i,:) = [iQStart, i, i+1];
    end
    Fface(n,:) = [iQStart, n, iQEnd];

    Fface = orientTriangles(Vface,Fface,desiredNormal);

    fprintf("Side normal [%g %g %g]: boundary points = %d, triangles = %d\n", ...
        desiredNormal(1),desiredNormal(2),desiredNormal(3),n,size(Fface,1));
end


function run = longestCircularRun(mask)
% Return indices of the longest TRUE run in a circular logical vector.

    mask = mask(:);
    n = numel(mask);

    if all(mask)
        run = (1:n).';
        return
    end

    doubled = [mask; mask];
    bestStart = [];
    bestLength = 0;
    i = 1;

    while i <= 2*n
        if doubled(i)
            j = i;
            while j <= 2*n && doubled(j)
                j = j + 1;
            end
            len = j-i;

            if len > bestLength && i <= n
                bestStart = i;
                bestLength = min(len,n);
            end
            i = j;
        else
            i = i+1;
        end
    end

    if isempty(bestStart)
        run = zeros(0,1);
        return
    end

    run = mod((bestStart:bestStart+bestLength-1)-1,n)+1;
    run = run(:);
end


function loop = orderSingleBoundaryLoop(E)
% Order an undirected edge list representing one closed loop.
%
% Returns vertex IDs once each; the first vertex is not repeated at the end.

    if isempty(E)
        error("The curved surface has no free boundary.");
    end

    G = graph(E(:,1),E(:,2));

    active = find(degree(G) > 0);
    if any(degree(G,active) ~= 2)
        error("Boundary is not one or more simple closed loops.");
    end

    bins = conncomp(G);
    activeBins = unique(bins(active));

    if numel(activeBins) ~= 1
        error("Expected one curved-face boundary loop, but found %d.",numel(activeBins));
    end

    start = active(1);
    loop = zeros(numel(active),1);
    loop(1) = start;

    prev = 0;
    curr = start;

    for k = 2:numel(active)
        nbr = neighbors(G,curr);

        if prev == 0
            next = nbr(1);
        else
            next = nbr(nbr ~= prev);
            if isempty(next)
                error("Boundary traversal failed.");
            end
            next = next(1);
        end

        loop(k) = next;
        prev = curr;
        curr = next;
    end

    if ~ismember(start,neighbors(G,curr))
        error("Boundary traversal did not close.");
    end
end


function F = orientTriangles(V,F,desiredNormal)
% Flip triangles so their normals have a positive dot product with
% desiredNormal.

    e1 = V(F(:,2),:) - V(F(:,1),:);
    e2 = V(F(:,3),:) - V(F(:,1),:);
    N = cross(e1,e2,2);

    flip = N*desiredNormal(:) < 0;
    F(flip,[2 3]) = F(flip,[3 2]);
end


function [Vall,Fall] = appendMesh(Vall,Fall,Vnew,Fnew)
    offset = size(Vall,1);
    Vall = [Vall; Vnew];
    Fall = [Fall; Fnew + offset];
end
