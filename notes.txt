function TRout = buildSlottedBlockSTL_v1(curvedStl,outStl, ...
    xLeft,xRight,yBottom,yTop,mergeTol)
%BUILDSLOTTEDBLOCKSTL_V1 Build a wider block with an open curved-bottom slot.
%
% The input STL is ONLY the finely triangulated curved slot-bottom surface.
% Material above this surface is removed up to yTop. The resulting solid:
%
%   - spans xLeft <= x <= xRight,
%   - spans yBottom <= y <= yTop,
%   - uses zMin/zMax from the curved STL,
%   - has an open slot between the curved surface's xMin and xMax,
%   - retains the original fine curved-surface triangulation,
%   - uses coarse triangulation on the remaining planar faces.
%
% Assumptions
% -----------
% 1. The curved STL contains one connected surface and one boundary loop.
% 2. Its X-Z projection has a rectangular outline.
% 3. The xMin/xMax boundary chains lie on constant-X planes.
% 4. The zMin/zMax boundary chains lie on constant-Z planes, while Y may
%    vary along those chains.
% 5. The remaining solid is below the curved surface; the removed slot is
%    above it.
%
% Example
% -------
% TR = buildSlottedBlockSTL_v1( ...
%     "curved_surface.stl","slotted_block.stl", ...
%     -20,40,-15,10,1e-7);

    arguments
        curvedStl (1,1) string
        outStl    (1,1) string
        xLeft     (1,1) double
        xRight    (1,1) double
        yBottom   (1,1) double
        yTop      (1,1) double
        mergeTol  (1,1) double = 1e-7
    end

    %% Read and inspect curved slot floor
    TRc = stlread(curvedStl);
    Vc = TRc.Points;
    Fc = TRc.ConnectivityList;

    Efree = freeBoundary(triangulation(Fc,Vc));
    loop = orderSingleBoundaryLoop(Efree);
    chains = splitLoopAtXZCorners(Vc,loop);

    Ploop = Vc(loop,:);
    xSlotMin = min(Ploop(:,1));
    xSlotMax = max(Ploop(:,1));
    zMin = min(Ploop(:,3));
    zMax = max(Ploop(:,3));

    if ~(xLeft < xSlotMin-mergeTol)
        error("xLeft must be less than the curved surface xMin.");
    end
    if ~(xRight > xSlotMax+mergeTol)
        error("xRight must be greater than the curved surface xMax.");
    end
    if ~(yBottom < min(Vc(:,2))-mergeTol)
        error("yBottom must be below every point on the curved surface.");
    end
    if ~(yTop > max(Vc(:,2))+mergeTol)
        error("yTop must be above every point on the curved surface.");
    end

    fprintf("Slot X range: [%g, %g]\n",xSlotMin,xSlotMax);
    fprintf("Block X range: [%g, %g]\n",xLeft,xRight);
    fprintf("Block Y range: [%g, %g]\n",yBottom,yTop);
    fprintf("Z range: [%g, %g]\n",zMin,zMax);
    fprintf("Boundary chain counts: xMin=%d, xMax=%d, zMin=%d, zMax=%d\n", ...
        numel(chains.xMin),numel(chains.xMax), ...
        numel(chains.zMin),numel(chains.zMax));

    %% Curved slot floor: outward normal points into removed material (+Y)
    Fc = orientTriangles(Vc,Fc,[0 1 0]);

    Vall = Vc;
    Fall = Fc;

    %% Slot vertical side walls
    % At slot xMin, remaining material is on the -X side, so outward is +X.
    [Vf,Ff] = makeSideFromChain(Vc,chains.xMin,yTop,1,[1 0 0]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    % At slot xMax, remaining material is on the +X side, so outward is -X.
    [Vf,Ff] = makeSideFromChain(Vc,chains.xMax,yTop,1,[-1 0 0]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    %% Top ledges on the two sides of the slot
    [Vf,Ff] = rectangleFace( ...
        [xLeft yTop zMin;
         xSlotMin yTop zMin;
         xSlotMin yTop zMax;
         xLeft yTop zMax], [0 1 0]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    [Vf,Ff] = rectangleFace( ...
        [xSlotMax yTop zMin;
         xRight yTop zMin;
         xRight yTop zMax;
         xSlotMax yTop zMax], [0 1 0]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    %% Outer X faces
    [Vf,Ff] = rectangleFace( ...
        [xLeft yBottom zMin;
         xLeft yTop    zMin;
         xLeft yTop    zMax;
         xLeft yBottom zMax], [-1 0 0]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    [Vf,Ff] = rectangleFace( ...
        [xRight yBottom zMin;
         xRight yTop    zMin;
         xRight yTop    zMax;
         xRight yBottom zMax], [1 0 0]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    %% Full bottom face
    [Vf,Ff] = rectangleFace( ...
        [xLeft  yBottom zMin;
         xRight yBottom zMin;
         xRight yBottom zMax;
         xLeft  yBottom zMax], [0 -1 0]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    %% Front and back end faces, including the curved slot profile
    [Vf,Ff] = makeSlottedEndFace(Vc,chains.zMin, ...
        xLeft,xRight,xSlotMin,xSlotMax,yBottom,yTop,zMin,[0 0 -1]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    [Vf,Ff] = makeSlottedEndFace(Vc,chains.zMax, ...
        xLeft,xRight,xSlotMin,xSlotMax,yBottom,yTop,zMax,[0 0 1]);
    [Vall,Fall] = appendMesh(Vall,Fall,Vf,Ff);

    %% Merge coincident vertices and remove duplicate/collapsed facets
    [Vout,~,map] = uniquetol(Vall,mergeTol, ...
        "ByRows",true,"DataScale",1);
    Fout = map(Fall);

    keep = Fout(:,1)~=Fout(:,2) & ...
           Fout(:,2)~=Fout(:,3) & ...
           Fout(:,3)~=Fout(:,1);
    Fout = Fout(keep,:);

    [~,ia] = unique(sort(Fout,2),"rows","stable");
    Fout = Fout(ia,:);

    TRout = triangulation(Fout,Vout);

    %% Topology and triangle checks
    reportMeshQuality(Vout,Fout);

    %% Write STL
    stlwrite(TRout,outStl);

    %% Plot
    figure("Name","Slotted block STL");
    trisurf(TRout, ...
        "FaceColor",[0.80 0.85 0.90], ...
        "EdgeColor",[0.20 0.20 0.20]);
    axis equal;
    grid on;
    xlabel("X"); ylabel("Y"); zlabel("Z");
    title("Block remaining after removing material above curved surface");
    view(3);
end


function [Vface,Fface] = makeSlottedEndFace(V,chainIds, ...
    xLeft,xRight,xSlotMin,xSlotMax,yBottom,yTop,zConst,normal)
% Create one constant-Z end face. Its upper boundary contains two horizontal
% ledges separated by the curved slot profile.

    chain = V(chainIds,:);

    % Snap Z only. Preserve the X-Y profile.
    chain(:,3) = zConst;

    % Remove near-duplicate consecutive vertices.
    s = max(max(abs(chain),[],"all"),1);
    keep = [true; vecnorm(diff(chain,1,1),2,2)>20*eps(s)];
    chain = chain(keep,:);

    % Order curved slot edge from xSlotMin to xSlotMax.
    if chain(1,1) > chain(end,1)
        chain = flipud(chain);
    end

    % Confirm endpoints correspond to slot sides.
    endTol = max(1e-8*max(xSlotMax-xSlotMin,1),100*eps(s));
    if abs(chain(1,1)-xSlotMin)>endTol || ...
       abs(chain(end,1)-xSlotMax)>endTol
        warning(["End-chain endpoints do not closely match slot X limits. " ...
                 "The topology check will reveal whether this is acceptable."]);
    end

    % Polygon in X-Y. Follow outer boundary counterclockwise:
    % lower-left -> lower-right -> upper-right -> right slot lip ->
    % curved slot floor right-to-left -> left slot lip -> upper-left.
    poly2 = [xLeft     yBottom;
             xRight    yBottom;
             xRight    yTop;
             xSlotMax  yTop;
             flipud(chain(:,[1 2]));
             xSlotMin  yTop;
             xLeft     yTop];

    poly2 = removeConsecutiveDuplicates(poly2);

    [P2,F2] = constrainedPolygonTriangulation(poly2);

    Vface = [P2(:,1),P2(:,2),repmat(zConst,size(P2,1),1)];
    Fface = orientTriangles(Vface,F2,normal);

    fprintf("End face Z=%g: %d curved-edge points, %d triangles\n", ...
        zConst,size(chain,1),size(Fface,1));
end


function [Vface,Fface] = makeSideFromChain(V,chainIds,yCap,constantAxis,normal)
% Create a planar slot wall from a complete curved boundary chain to yCap.

    chain = V(chainIds,:);

    s = max(max(abs(chain),[],"all"),1);
    keep = [true; vecnorm(diff(chain,1,1),2,2)>20*eps(s)];
    chain = chain(keep,:);

    if constantAxis==1
        chain(:,1) = mean(chain(:,1));
        poly3 = [chain;
                 setY(chain(end,:),yCap);
                 setY(chain(1,:),yCap)];
        poly2 = poly3(:,[2 3]);
    elseif constantAxis==3
        chain(:,3) = mean(chain(:,3));
        poly3 = [chain;
                 setY(chain(end,:),yCap);
                 setY(chain(1,:),yCap)];
        poly2 = poly3(:,[1 2]);
    else
        error("constantAxis must be 1 or 3.");
    end

    poly2 = removeConsecutiveDuplicates(poly2);
    [P2,F2] = constrainedPolygonTriangulation(poly2);

    Vface = zeros(size(P2,1),3);
    if constantAxis==1
        Vface(:,1)=mean(chain(:,1));
        Vface(:,2)=P2(:,1);
        Vface(:,3)=P2(:,2);
    else
        Vface(:,1)=P2(:,1);
        Vface(:,2)=P2(:,2);
        Vface(:,3)=mean(chain(:,3));
    end

    Fface = orientTriangles(Vface,F2,normal);

    fprintf("Slot wall [%g %g %g]: %d boundary points, %d triangles\n", ...
        normal(1),normal(2),normal(3),size(chain,1),size(Fface,1));
end


function p = setY(p,y)
    p(2)=y;
end


function [P2,F2] = constrainedPolygonTriangulation(poly2)
% Triangulate a simple 2-D polygon while retaining all supplied boundary
% vertices and discarding triangles outside a concave polygon.

    poly2 = removeConsecutiveDuplicates(poly2);

    if size(poly2,1)<3
        error("Polygon has fewer than three distinct vertices.");
    end

    % Avoid repeating the first point at the end.
    if norm(poly2(1,:)-poly2(end,:)) <= ...
            20*eps(max(max(abs(poly2),[],"all"),1))
        poly2(end,:)=[];
    end

    n=size(poly2,1);
    C=[(1:n-1)' (2:n)'; n 1];

    DT=delaunayTriangulation(poly2,C);
    Fcand=DT.ConnectivityList;
    P2=DT.Points;

    ctr=(P2(Fcand(:,1),:)+P2(Fcand(:,2),:)+P2(Fcand(:,3),:))/3;
    [in,on]=inpolygon(ctr(:,1),ctr(:,2),poly2(:,1),poly2(:,2));
    F2=Fcand(in|on,:);

    if isempty(F2)
        error("Polygon triangulation produced no interior triangles.");
    end
end


function P = removeConsecutiveDuplicates(P)
    if isempty(P)
        return
    end
    s=max(max(abs(P),[],"all"),1);
    keep=[true;vecnorm(diff(P,1,1),2,2)>20*eps(s)];
    P=P(keep,:);
end


function [Vface,Fface] = rectangleFace(Vface,normal)
    Fface=orientTriangles(Vface,[1 2 3;1 3 4],normal);
end


function chains = splitLoopAtXZCorners(V,loop)
% Find the four geometric corners in normalized X-Z projection and split
% the ordered loop into four complete chains.

    P=V(loop,:);
    x=P(:,1);
    z=P(:,3);

    x0=min(x); x1=max(x);
    z0=min(z); z1=max(z);
    dx=x1-x0; dz=z1-z0;

    if dx<=0 || dz<=0
        error("Boundary has zero X or Z extent.");
    end

    xn=(x-x0)/dx;
    zn=(z-z0)/dz;

    targets=[0 0;1 0;1 1;0 1];
    idx=zeros(4,1);

    for k=1:4
        d2=(xn-targets(k,1)).^2+(zn-targets(k,2)).^2;
        [~,idx(k)]=min(d2);
    end

    if numel(unique(idx))~=4
        error("Could not identify four distinct X-Z boundary corners.");
    end

    idxSorted=sort(idx);
    C=cell(4,1);
    n=numel(loop);

    for k=1:4
        i1=idxSorted(k);
        if k<4
            i2=idxSorted(k+1);
            pos=i1:i2;
        else
            i2=idxSorted(1);
            pos=[i1:n,1:i2];
        end
        C{k}=loop(pos(:));
    end

    meanX=zeros(4,1); meanZ=zeros(4,1);
    rangeX=zeros(4,1); rangeZ=zeros(4,1);

    for k=1:4
        Q=V(C{k},:);
        meanX(k)=mean(Q(:,1));
        meanZ(k)=mean(Q(:,3));
        rangeX(k)=(max(Q(:,1))-min(Q(:,1)))/dx;
        rangeZ(k)=(max(Q(:,3))-min(Q(:,3)))/dz;
    end

    xCandidates=find(rangeX<=rangeZ);
    zCandidates=find(rangeZ<rangeX);

    if numel(xCandidates)~=2 || numel(zCandidates)~=2
        score=rangeX-rangeZ;
        [~,ord]=sort(score);
        xCandidates=ord(1:2);
        zCandidates=ord(3:4);
    end

    [~,a]=min(meanX(xCandidates));
    [~,b]=max(meanX(xCandidates));
    iXMin=xCandidates(a);
    iXMax=xCandidates(b);

    [~,a]=min(meanZ(zCandidates));
    [~,b]=max(meanZ(zCandidates));
    iZMin=zCandidates(a);
    iZMax=zCandidates(b);

    chains.xMin=C{iXMin};
    chains.xMax=C{iXMax};
    chains.zMin=C{iZMin};
    chains.zMax=C{iZMax};
end


function loop = orderSingleBoundaryLoop(E)
    if isempty(E)
        error("The curved surface has no free boundary.");
    end

    G=graph(E(:,1),E(:,2));
    active=find(degree(G)>0);

    if any(degree(G,active)~=2)
        error("The free boundary is not a simple closed loop.");
    end

    comp=conncomp(G);
    if numel(unique(comp(active)))~=1
        error("Expected exactly one free-boundary loop.");
    end

    loop=zeros(numel(active),1);
    loop(1)=active(1);
    prev=0;
    curr=loop(1);

    for k=2:numel(active)
        nb=neighbors(G,curr);
        if prev==0
            nxt=nb(1);
        else
            nxt=nb(nb~=prev);
            if isempty(nxt)
                error("Boundary traversal failed.");
            end
            nxt=nxt(1);
        end
        loop(k)=nxt;
        prev=curr;
        curr=nxt;
    end

    if ~ismember(loop(1),neighbors(G,curr))
        error("Boundary traversal did not close.");
    end
end


function reportMeshQuality(V,F)
    E=[F(:,[1 2]);F(:,[2 3]);F(:,[3 1])];
    E=sort(E,2);
    [~,~,ie]=unique(E,"rows");
    count=accumarray(ie,1);

    nOpen=nnz(count==1);
    nNonmanifold=nnz(count>2);

    e1=V(F(:,2),:)-V(F(:,1),:);
    e2=V(F(:,3),:)-V(F(:,1),:);
    twiceArea=vecnorm(cross(e1,e2,2),2,2);

    scale=max(max(V,[],1)-min(V,[],1));
    areaTol=1e-12*max(scale,1)^2;
    nCollapsed=nnz(0.5*twiceArea<areaTol);

    fprintf("Output vertices:        %d\n",size(V,1));
    fprintf("Output triangles:       %d\n",size(F,1));
    fprintf("Open edges:             %d\n",nOpen);
    fprintf("Nonmanifold edges:      %d\n",nNonmanifold);
    fprintf("Near-collapsed triangles:%d\n",nCollapsed);

    if nOpen~=0 || nNonmanifold~=0 || nCollapsed~=0
        warning("Mesh checks found problems; inspect the plotted STL.");
    end
end


function F = orientTriangles(V,F,desiredNormal)
    e1=V(F(:,2),:)-V(F(:,1),:);
    e2=V(F(:,3),:)-V(F(:,1),:);
    N=cross(e1,e2,2);
    flip=N*desiredNormal(:)<0;
    F(flip,[2 3])=F(flip,[3 2]);
end


function [Vall,Fall] = appendMesh(Vall,Fall,Vnew,Fnew)
    off=size(Vall,1);
    Vall=[Vall;Vnew];
    Fall=[Fall;Fnew+off];
end
